cmake_minimum_required(VERSION 3.16)
project(Frond VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
include_directories(../build/source/src/common)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
add_library(frond-client STATIC DesktopClient.cpp)
target_link_libraries(frond-client PUBLIC Qt6::Network)
target_include_directories(frond-client PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(frond-sd-exchange STATIC SdExchange.cpp SdExchangeDialog.cpp)
target_link_libraries(frond-sd-exchange PUBLIC Qt6::Widgets)
target_include_directories(frond-sd-exchange PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(frond-desktop main.cpp MainWindow.cpp)
target_link_libraries(frond-desktop PRIVATE frond-client frond-sd-exchange Qt6::Widgets)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(sd-exchange-driver ../tests/sd-exchange-driver.cpp)
  target_link_libraries(sd-exchange-driver PRIVATE frond-sd-exchange)
  add_executable(test-sd-exchange ../tests/sd-exchange.cpp)
  target_include_directories(test-sd-exchange PRIVATE ../build/source/src/uarm/native)
  target_link_libraries(test-sd-exchange PRIVATE frond-sd-exchange Qt6::Test)
  add_test(NAME sd-exchange COMMAND test-sd-exchange)
  add_executable(test-firmware-input ../tests/firmware-input.cpp)
  target_include_directories(test-firmware-input PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(test-firmware-input PRIVATE Qt6::Test)
  add_test(NAME firmware-input COMMAND test-firmware-input)
  add_executable(test-desktop-client ../tests/desktop-client.cpp)
  target_link_libraries(test-desktop-client PRIVATE frond-client Qt6::Test)
  add_test(NAME desktop-client COMMAND test-desktop-client)
  add_executable(test-desktop-workflow ../tests/desktop-workflow.cpp MainWindow.cpp)
  target_link_libraries(test-desktop-workflow PRIVATE frond-client frond-sd-exchange Qt6::Widgets Qt6::Test)
  target_include_directories(test-desktop-workflow PRIVATE ../build/source/src/uarm/native)
  add_test(NAME desktop-sd-exchange COMMAND test-desktop-workflow sdExchange)
  set_tests_properties(desktop-sd-exchange PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME desktop-save-failure COMMAND test-desktop-workflow failedSaveStaysOpen)
  set_tests_properties(desktop-save-failure PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS frond-desktop RUNTIME DESTINATION bin)
