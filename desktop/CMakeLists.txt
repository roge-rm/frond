cmake_minimum_required(VERSION 3.16)
file(STRINGS ${CMAKE_CURRENT_SOURCE_DIR}/../VERSION FROND_VERSION_STRING LIMIT_COUNT 1)
project(Frond VERSION ${FROND_VERSION_STRING} LANGUAGES C CXX)
add_compile_definitions(FROND_VERSION="${FROND_VERSION_STRING}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../VERSION)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(FROND_PREPARED_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../build/source" CACHE PATH "Prepared patched core source")
include_directories(${FROND_PREPARED_SOURCE}/src/common ${FROND_PREPARED_SOURCE}/src/uarm/native)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network PrintSupport)
find_package(ZLIB REQUIRED)
# xcb sets the X input focus on the embedded Palm window.
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(XCB IMPORTED_TARGET xcb)
endif()
# The link protocol (../link) with Monocypher, the hub and the window side of it.
add_library(frond-linkproto STATIC ../link/crypto.cpp ../link/noise.cpp ../link/wire.cpp ../link/pairing.cpp
  ../link/messages.cpp ../link/mux.cpp ../link/link.cpp ../link/monocypher/monocypher.c)
set_target_properties(frond-linkproto PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE ON)
add_library(frond-hub STATIC LinkStore.cpp LinkHub.cpp LinkClient.cpp ConnectionsWindow.cpp)
target_link_libraries(frond-hub PUBLIC frond-linkproto Qt6::Network Qt6::Widgets)
target_include_directories(frond-hub PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(frond-client STATIC DesktopClient.cpp DanaViewController.cpp)
target_link_libraries(frond-client PUBLIC Qt6::Network)
target_include_directories(frond-client PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
# Card files edited in place (the storage library through CardStorage.h, the file manager's operations).
add_library(frond-card-files STATIC CardFiles.cpp ${FROND_PREPARED_SOURCE}/src/common/zip/miniz.c)
target_link_libraries(frond-card-files PUBLIC Qt6::Core ZLIB::ZLIB)
target_include_directories(frond-card-files PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(frond-sd-exchange STATIC SdExchange.cpp)
target_link_libraries(frond-sd-exchange PUBLIC frond-card-files Qt6::Core)
target_include_directories(frond-sd-exchange PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(frond-dialogs STATIC Dialogs.cpp DanaViewControls.cpp)
target_link_libraries(frond-dialogs PUBLIC Qt6::Widgets Qt6::Network ZLIB::ZLIB)
target_include_directories(frond-dialogs PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
# The USB printer row and drawing a received job on a printer or PDF.
add_library(frond-print STATIC PrintJob.cpp PrinterPanel.cpp)
target_link_libraries(frond-print PUBLIC Qt6::Widgets Qt6::PrintSupport)
target_include_directories(frond-print PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../print)
add_executable(frond-desktop main.cpp MainWindow.cpp RunningBar.cpp RunningPanels.cpp PalmView.cpp CardsWindow.cpp CardFilesWindow.cpp LicencesDialog.cpp licences.qrc)
target_link_libraries(frond-desktop PRIVATE frond-client frond-card-files frond-dialogs frond-hub frond-print Qt6::Widgets)
if(TARGET PkgConfig::XCB)
  target_link_libraries(frond-desktop PRIVATE PkgConfig::XCB)
  target_compile_definitions(frond-desktop PRIVATE FROND_XCB)
endif()
