cmake_minimum_required(VERSION 3.16)
project(Frond VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
add_library(frond-client STATIC DesktopClient.cpp)
target_link_libraries(frond-client PUBLIC Qt6::Network)
target_include_directories(frond-client PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(frond-desktop main.cpp MainWindow.cpp)
target_link_libraries(frond-desktop PRIVATE frond-client Qt6::Widgets)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(test-desktop-client ../tests/desktop-client.cpp)
  target_link_libraries(test-desktop-client PRIVATE frond-client Qt6::Test)
  add_test(NAME desktop-client COMMAND test-desktop-client)
  add_executable(test-desktop-workflow ../tests/desktop-workflow.cpp MainWindow.cpp)
  target_link_libraries(test-desktop-workflow PRIVATE frond-client Qt6::Widgets Qt6::Test)
  add_test(NAME desktop-save-failure COMMAND test-desktop-workflow failedSaveStaysOpen)
  set_tests_properties(desktop-save-failure PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS frond-desktop RUNTIME DESTINATION bin)
